import type { GameAssets, Atlas } from "./assets";
import { CELL, ENEMIES, TOWERS, WORLD_H, WORLD_W, towerCombat } from "./config";
import type { Game } from "./engine";
import { getWorldTheme } from "./evolution";
import { cellCenter, inBounds } from "./map";
import type { Enemy, Tower, TowerBranchId, TowerKind } from "./types";

export type View = { scale: number; ox: number; oy: number; cssW: number; cssH: number };

export function renderFrame(
  ctx: CanvasRenderingContext2D,
  game: Game,
  assets: GameAssets,
  view: View,
  time: number,
  shakeX: number,
  shakeY: number,
  mapCanvas: HTMLCanvasElement | null,
) {
  ctx.clearRect(0, 0, view.cssW, view.cssH);
  ctx.fillStyle = "#0c0b0a";
  ctx.fillRect(0, 0, view.cssW, view.cssH);

  ctx.save();
  ctx.translate(view.ox + shakeX, view.oy + shakeY);
  ctx.scale(view.scale, view.scale);

  if (mapCanvas) ctx.drawImage(mapCanvas, 0, 0);
  drawEraTerrain(ctx, game, time);
  drawHover(ctx, game, assets);
  drawRift(ctx, game, time, game.eraIndex);
  drawKeep(ctx, game, assets, time, game.eraIndex);
  drawRange(ctx, game);
  drawEnemies(ctx, game, assets);
  drawTowers(ctx, game, assets);
  drawProjectiles(ctx, game);
  drawParticles(ctx, game);
  drawFloats(ctx, game);
  drawEraAmbient(ctx, game.eraIndex, time);
  drawVignette(ctx);

  ctx.restore();
}

function drawHover(ctx: CanvasRenderingContext2D, game: Game, assets: GameAssets) {
  const { hoverCol: col, hoverRow: row } = game;
  if (!inBounds(col, row)) return;
  if (game.phase === "menu") return;
  const can = game.canBuild(col, row);
  const placing = Boolean(game.selectedKind);
  ctx.save();
  ctx.globalAlpha = placing ? 0.55 : 0.28;
  ctx.drawImage(assets.pad, col * CELL + 3, row * CELL + 3, CELL - 6, CELL - 6);
  ctx.restore();
  ctx.strokeStyle = can && placing ? "rgba(216,210,200,0.7)" : "rgba(181,74,60,0.55)";
  ctx.lineWidth = 2;
  ctx.strokeRect(col * CELL + 6, row * CELL + 6, CELL - 12, CELL - 12);
  if (placing && game.selectedKind && can) {
    const img = assets[game.selectedKind];
    ctx.globalAlpha = 0.7;
    ctx.drawImage(img, col * CELL + 6, row * CELL - 8, CELL - 12, CELL - 4);
    ctx.globalAlpha = 1;
  }
}

function drawEraTerrain(ctx: CanvasRenderingContext2D, game: Game, time: number) {
  const era = Math.max(0, Math.min(4, game.eraIndex));
  const theme = getWorldTheme(era);
  const palette = [
    { ground: "rgba(54,34,27,0.20)", edge: "rgba(42,25,19,0.72)", path: "rgba(99,58,39,0.48)", trim: "rgba(196,92,38,0.24)" },
    { ground: "rgba(103,67,31,0.58)", edge: "rgba(57,36,19,0.96)", path: "rgba(155,105,52,0.90)", trim: "rgba(238,190,101,0.72)" },
    { ground: "rgba(52,58,64,0.62)", edge: "rgba(28,31,34,0.96)", path: "rgba(105,113,120,0.92)", trim: "rgba(199,207,214,0.68)" },
    { ground: "rgba(29,35,39,0.70)", edge: "rgba(17,20,22,0.98)", path: "rgba(70,76,79,0.96)", trim: "rgba(208,76,54,0.74)" },
    { ground: "rgba(16,38,57,0.72)", edge: "rgba(8,22,34,0.98)", path: "rgba(31,79,101,0.94)", trim: "rgba(120,225,255,0.84)" },
  ][era];

  ctx.save();
  ctx.fillStyle = palette.ground;
  ctx.fillRect(0, 0, WORLD_W, WORLD_H);
  ctx.fillStyle = theme.tint;
  ctx.fillRect(0, 0, WORLD_W, WORLD_H);
  drawEraGroundIdentity(ctx, era, time);

  const tracePath = () => {
    ctx.beginPath();
    game.path.points.forEach((p, i) => (i === 0 ? ctx.moveTo(p.x, p.y) : ctx.lineTo(p.x, p.y)));
  };

  ctx.lineJoin = "round";
  ctx.lineCap = "round";
  ctx.strokeStyle = palette.edge;
  ctx.lineWidth = CELL * 0.92;
  tracePath();
  ctx.stroke();

  ctx.strokeStyle = palette.path;
  ctx.lineWidth = CELL * 0.68;
  tracePath();
  ctx.stroke();

  ctx.strokeStyle = palette.trim;
  ctx.lineWidth = era === 0 ? 2 : era === 4 ? 4 : 3;
  if (era === 1) ctx.setLineDash([18, 12]);
  if (era === 2) ctx.setLineDash([12, 8]);
  if (era === 3) ctx.setLineDash([22, 7, 4, 7]);
  if (era === 4) ctx.setLineDash([5, 10]);
  ctx.lineDashOffset = -time * (era === 4 ? 30 : era >= 2 ? 9 : 4);
  tracePath();
  ctx.stroke();
  ctx.setLineDash([]);

  drawEraProps(ctx, era, time);
  ctx.restore();
}

function drawEraGroundIdentity(ctx: CanvasRenderingContext2D, era: number, time: number) {
  ctx.save();

  if (era === 0) {
    ctx.strokeStyle = "rgba(116,78,56,0.10)";
    ctx.lineWidth = 1;
    for (let y = 22; y < WORLD_H; y += 58) {
      ctx.beginPath();
      ctx.moveTo(0, y + Math.sin(y * 0.03) * 5);
      ctx.lineTo(WORLD_W, y + Math.sin(y * 0.03 + 1.4) * 5);
      ctx.stroke();
    }
  } else if (era === 1) {
    ctx.fillStyle = "rgba(215,165,84,0.10)";
    for (let y = 0; y < WORLD_H; y += 52) {
      for (let x = (Math.floor(y / 52) % 2) * 26; x < WORLD_W; x += 52) {
        ctx.fillRect(x + 2, y + 2, 48, 48);
      }
    }
    ctx.strokeStyle = "rgba(245,204,120,0.25)";
    ctx.lineWidth = 2;
    for (let x = -WORLD_H; x < WORLD_W; x += 92) {
      ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x + WORLD_H, WORLD_H); ctx.stroke();
    }
    ctx.fillStyle = "rgba(112,67,27,0.30)";
    for (let x = 44; x < WORLD_W; x += 154) for (let y = 38; y < WORLD_H; y += 132) {
      ctx.beginPath();
      ctx.moveTo(x, y - 9); ctx.lineTo(x + 9, y); ctx.lineTo(x, y + 9); ctx.lineTo(x - 9, y); ctx.closePath();
      ctx.fill();
    }
  } else if (era === 2) {
    ctx.strokeStyle = "rgba(210,217,222,0.24)";
    ctx.lineWidth = 2;
    for (let x = 0; x <= WORLD_W; x += CELL) {
      ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, WORLD_H); ctx.stroke();
    }
    for (let y = 0; y <= WORLD_H; y += CELL) {
      ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(WORLD_W, y); ctx.stroke();
    }
    ctx.fillStyle = "rgba(24,28,31,0.32)";
    for (let x = CELL * 0.5; x < WORLD_W; x += CELL) for (let y = CELL * 0.5; y < WORLD_H; y += CELL) {
      ctx.beginPath(); ctx.arc(x, y, 3, 0, Math.PI * 2); ctx.fill();
    }
  } else if (era === 3) {
    ctx.strokeStyle = "rgba(154,164,168,0.22)";
    ctx.lineWidth = 5;
    for (let x = 28; x < WORLD_W; x += 126) {
      ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, WORLD_H); ctx.stroke();
    }
    ctx.strokeStyle = "rgba(203,77,55,0.28)";
    ctx.lineWidth = 3;
    for (let y = 36; y < WORLD_H; y += 92) {
      ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(WORLD_W, y); ctx.stroke();
    }
    ctx.fillStyle = "rgba(232,126,66,0.26)";
    for (let i = 0; i < 22; i++) {
      const x = (i * 109) % WORLD_W;
      const y = (i * 67) % WORLD_H;
      ctx.fillRect(x, y, 11, 3);
    }
  } else {
    const drift = (time * 18) % CELL;
    ctx.strokeStyle = "rgba(122,226,255,0.26)";
    ctx.lineWidth = 1.2;
    for (let x = -CELL + drift; x < WORLD_W + CELL; x += CELL) {
      ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, WORLD_H); ctx.stroke();
    }
    for (let y = 0; y < WORLD_H; y += CELL) {
      ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(WORLD_W, y); ctx.stroke();
    }
    ctx.fillStyle = "rgba(134,226,255,0.28)";
    for (let i = 0; i < 24; i++) {
      const x = (i * 127 + time * 24) % WORLD_W;
      const y = (i * 83) % WORLD_H;
      ctx.fillRect(x, y, 16, 2);
      ctx.fillRect(x + 14, y, 2, 12);
    }
  }

  ctx.restore();
}

function drawEraProps(ctx: CanvasRenderingContext2D, era: number, time: number) {
  const points = [
    { x: 5.4 * CELL, y: 0.55 * CELL },
    { x: 12.5 * CELL, y: 2.5 * CELL },
    { x: 0.72 * CELL, y: 6.8 * CELL },
    { x: 7.0 * CELL, y: 7.75 * CELL },
  ];

  ctx.save();
  if (era === 0) {
    ctx.fillStyle = "rgba(72,55,45,0.48)";
    for (const p of points) {
      ctx.beginPath(); ctx.arc(p.x, p.y, 10, 0, Math.PI * 2); ctx.fill();
      ctx.fillRect(p.x - 3, p.y - 18, 6, 19);
    }
  } else if (era === 1) {
    for (const p of points) {
      ctx.fillStyle = "rgba(70,43,22,0.88)";
      ctx.fillRect(p.x - 13, p.y - 12, 26, 28);
      ctx.fillStyle = "rgba(190,126,55,0.92)";
      ctx.fillRect(p.x - 10, p.y - 9, 20, 22);
      ctx.strokeStyle = "rgba(245,202,111,0.92)";
      ctx.lineWidth = 3;
      ctx.beginPath(); ctx.moveTo(p.x, p.y - 32); ctx.lineTo(p.x, p.y + 14); ctx.stroke();
      ctx.fillStyle = "rgba(179,86,40,0.95)";
      ctx.beginPath(); ctx.moveTo(p.x + 2, p.y - 29); ctx.lineTo(p.x + 23, p.y - 20); ctx.lineTo(p.x + 2, p.y - 11); ctx.closePath(); ctx.fill();
      ctx.fillStyle = "rgba(247,211,129,0.80)";
      ctx.beginPath(); ctx.arc(p.x, p.y - 2, 5, 0, Math.PI * 2); ctx.fill();
    }
  } else if (era === 2) {
    for (const p of points) {
      ctx.fillStyle = "rgba(45,49,53,0.90)"; ctx.fillRect(p.x - 18, p.y - 14, 36, 28);
      ctx.strokeStyle = "rgba(199,207,214,0.88)"; ctx.lineWidth = 3; ctx.strokeRect(p.x - 18, p.y - 14, 36, 28);
      ctx.fillStyle = "rgba(210,146,67,0.90)";
      for (const sx of [-12, 12]) for (const sy of [-8, 8]) { ctx.beginPath(); ctx.arc(p.x + sx, p.y + sy, 2.5, 0, Math.PI * 2); ctx.fill(); }
    }
  } else if (era === 3) {
    for (const p of points) {
      ctx.fillStyle = "rgba(38,43,46,0.94)"; ctx.fillRect(p.x - 21, p.y - 13, 42, 26);
      ctx.strokeStyle = "rgba(164,174,178,0.84)"; ctx.lineWidth = 6;
      ctx.beginPath(); ctx.moveTo(p.x - 25, p.y + 17); ctx.lineTo(p.x + 18, p.y + 17); ctx.lineTo(p.x + 18, p.y - 22); ctx.stroke();
      ctx.fillStyle = "rgba(210,75,54,0.94)"; ctx.fillRect(p.x - 26, p.y + 11, 10, 12);
      ctx.fillStyle = "rgba(232,151,69,0.86)"; ctx.fillRect(p.x + 8, p.y - 7, 7, 7);
    }
  } else {
    for (let i = 0; i < points.length; i++) {
      const p = points[i];
      const pulse = 0.62 + Math.sin(time * 2.4 + i) * 0.18;
      ctx.strokeStyle = `rgba(134,226,255,${pulse})`;
      ctx.lineWidth = 2.5;
      ctx.beginPath(); ctx.arc(p.x, p.y, 16 + Math.sin(time + i) * 2, 0, Math.PI * 2); ctx.stroke();
      ctx.beginPath(); ctx.arc(p.x, p.y, 27 + Math.cos(time * 1.4 + i) * 2, 0, Math.PI * 2); ctx.stroke();
      ctx.fillStyle = "rgba(134,226,255,0.48)";
      ctx.fillRect(p.x - 4, p.y - 4, 8, 8);
      ctx.beginPath(); ctx.moveTo(p.x - 34, p.y); ctx.lineTo(p.x - 17, p.y); ctx.moveTo(p.x + 17, p.y); ctx.lineTo(p.x + 34, p.y); ctx.stroke();
    }
  }
  ctx.restore();
}

function drawRift(ctx: CanvasRenderingContext2D, game: Game, time: number, eraIndex: number) {
  const p = cellCenter(game.spawnCell[0], game.spawnCell[1]);
  const theme = getWorldTheme(eraIndex);
  ctx.save();
  ctx.translate(p.x, p.y);
  for (let i = 0; i < 3; i++) {
    ctx.beginPath();
    ctx.globalAlpha = 0.58 - i * 0.13;
    ctx.strokeStyle = theme.riftColor;
    ctx.lineWidth = 2.5 - i * 0.4;
    ctx.arc(0, 0, 16 + i * 7 + Math.sin(time * 2.2 + i) * 2.2, 0, Math.PI * 2);
    ctx.stroke();
  }
  ctx.globalAlpha = 0.38;
  ctx.fillStyle = theme.riftColor;
  ctx.beginPath();
  ctx.arc(0, 0, 8 + Math.sin(time * 3) * 1.5, 0, Math.PI * 2);
  ctx.fill();
  if (eraIndex === 4) {
    ctx.globalAlpha = 0.55; ctx.rotate(time * 0.3); ctx.strokeStyle = theme.riftColor; ctx.lineWidth = 1.5; ctx.strokeRect(-25, -25, 50, 50);
  }
  ctx.restore();
}

function drawKeep(ctx: CanvasRenderingContext2D, game: Game, assets: GameAssets, time: number, eraIndex: number) {
  const p = cellCenter(game.keepCell[0], game.keepCell[1]);
  const theme = getWorldTheme(eraIndex);
  const s = CELL * 1.35;
  ctx.save();
  if (eraIndex >= 1) {
    ctx.globalAlpha = 0.24 + eraIndex * 0.05; ctx.strokeStyle = theme.keepAccent; ctx.lineWidth = 3;
    ctx.beginPath(); ctx.arc(p.x, p.y - 6, 34 + eraIndex * 3 + Math.sin(time * 1.7) * 1.5, 0, Math.PI * 2); ctx.stroke();
  }
  if (eraIndex === 2) { ctx.fillStyle = "rgba(125,132,138,0.55)"; ctx.fillRect(p.x - 48, p.y + 18, 96, 12); }
  if (eraIndex === 3) { ctx.strokeStyle = "rgba(150,155,158,0.55)"; ctx.lineWidth = 6; ctx.beginPath(); ctx.moveTo(p.x - 48, p.y + 25); ctx.lineTo(p.x + 48, p.y + 25); ctx.stroke(); }
  if (eraIndex === 4) {
    ctx.globalAlpha = 0.35; ctx.strokeStyle = theme.keepAccent; ctx.lineWidth = 2;
    for (let r = 40; r <= 52; r += 12) { ctx.beginPath(); ctx.arc(p.x, p.y - 5, r + Math.sin(time * 2 + r) * 2, 0, Math.PI * 2); ctx.stroke(); }
  }
  ctx.globalAlpha = 1;
  ctx.drawImage(assets.keep, p.x - s / 2, p.y - s * 0.72, s, s);
  ctx.fillStyle = theme.keepAccent; ctx.globalAlpha = 0.38 + Math.sin(time * 3) * 0.12;
  ctx.beginPath(); ctx.arc(p.x + 6, p.y - 10, 3 + eraIndex * 0.4, 0, Math.PI * 2); ctx.fill();
  ctx.restore();
}

function drawRange(ctx: CanvasRenderingContext2D, game: Game) {
  const t =
    game.towers.find((x) => x.id === game.selectedId) ??
    (game.selectedKind && inBounds(game.hoverCol, game.hoverRow) && game.canBuild(game.hoverCol, game.hoverRow)
      ? ({
          x: cellCenter(game.hoverCol, game.hoverRow).x,
          y: cellCenter(game.hoverCol, game.hoverRow).y,
          kind: game.selectedKind,
          dmgLevel: 0,
          rateLevel: 0,
        } as Pick<Tower, "x" | "y" | "kind" | "dmgLevel" | "rateLevel">)
      : null);
  if (!t) return;
  const range = towerCombat(t.kind, t.dmgLevel, t.rateLevel, game.towerTechEra(t.kind), game.towerBranch(t.kind)).range;
  ctx.beginPath();
  ctx.arc(t.x, t.y, range, 0, Math.PI * 2);
  ctx.fillStyle = "rgba(216,210,200,0.07)";
  ctx.fill();
  ctx.strokeStyle = "rgba(216,210,200,0.35)";
  ctx.lineWidth = 1.5;
  ctx.stroke();
}

function drawTowers(ctx: CanvasRenderingContext2D, game: Game, assets: GameAssets) {
  for (const t of game.towers) {
    const img = assets[t.kind];
    const rec = t.recoil * 3;
    const size = CELL * (t.kind === "spire" ? 1.18 : 1.05);
    ctx.save();
    ctx.translate(t.x, t.y);
    ctx.drawImage(img, -size / 2, -size / 2 - 8 - rec, size, size);
    ctx.restore();
    drawTowerEvolution(ctx, t, game.towerTechEra(t.kind), game.towerBranch(t.kind));
    if (game.selectedId === t.id) {
      ctx.strokeStyle = "rgba(216,210,200,0.85)";
      ctx.lineWidth = 2;
      ctx.strokeRect(t.x - CELL / 2 + 4, t.y - CELL / 2 + 4, CELL - 8, CELL - 8);
    }
  }
}

function drawTowerEvolution(ctx: CanvasRenderingContext2D, t: Tower, era: number, branch: TowerBranchId | null) {
  if (era <= 0) return;
  ctx.save();
  ctx.translate(t.x, t.y - 8);
  ctx.globalAlpha = 0.72;
  if (era === 1) {
    ctx.strokeStyle = "#c4b08a"; ctx.lineWidth = 3; ctx.strokeRect(-18, -18, 36, 36);
    ctx.fillStyle = "rgba(196,176,138,0.28)";
    if (t.kind === "ballista") { ctx.fillRect(-28, -4, 56, 8); }
    if (t.kind === "mortar") { ctx.beginPath(); ctx.arc(0, 0, 16, 0, Math.PI * 2); ctx.fill(); }
    if (t.kind === "spire") { ctx.beginPath(); ctx.moveTo(0, -28); ctx.lineTo(-15, 18); ctx.lineTo(15, 18); ctx.closePath(); ctx.fill(); }
  } else if (era === 2) {
    ctx.strokeStyle = "#8f8a82"; ctx.lineWidth = 3; ctx.beginPath(); ctx.arc(0, 0, 23, 0, Math.PI * 2); ctx.stroke();
    ctx.fillStyle = "rgba(143,138,130,0.24)";
    ctx.fillRect(-24, -22, 8, 44); ctx.fillRect(16, -22, 8, 44);
  } else if (era === 3) {
    ctx.strokeStyle = "#b54a3c"; ctx.lineWidth = 4; ctx.beginPath(); ctx.arc(0, 0, 25, 0, Math.PI * 2); ctx.stroke();
    ctx.fillStyle = "rgba(181,74,60,0.24)";
    ctx.fillRect(-29, 14, 58, 8);
    ctx.strokeStyle = "#8f8a82"; ctx.lineWidth = 3;
    ctx.beginPath(); ctx.moveTo(-26, -18); ctx.lineTo(26, -18); ctx.stroke();
    if (t.kind === "ballista") { ctx.fillStyle = "#c4b08a"; ctx.fillRect(-4, -31, 8, 12); ctx.fillRect(-4, 19, 8, 12); }
  } else {
    const g = ctx.createRadialGradient(0, 0, 5, 0, 0, 34); g.addColorStop(0, "rgba(134,214,239,0.62)"); g.addColorStop(1, "rgba(134,214,239,0)");
    ctx.fillStyle = g; ctx.beginPath(); ctx.arc(0, 0, 34, 0, Math.PI * 2); ctx.fill();
    ctx.strokeStyle = "#d8f3ff"; ctx.lineWidth = 2; ctx.beginPath(); ctx.arc(0, 0, 28, 0, Math.PI * 2); ctx.stroke();
    ctx.rotate(Math.PI / 4); ctx.strokeRect(-21, -21, 42, 42);
  }
  ctx.globalAlpha = 1;
  drawBranchSignature(ctx, branch);
  ctx.restore();
}

function drawBranchSignature(ctx: CanvasRenderingContext2D, branch: TowerBranchId | null) {
  if (!branch) return;
  ctx.lineWidth = 2.5;
  if (branch === "ballista:siege") {
    ctx.strokeStyle = "#d8c39b";
    ctx.strokeRect(-26, -5, 52, 10);
  } else if (branch === "ballista:rapid") {
    ctx.strokeStyle = "#e6dfd4";
    for (let x = -20; x <= 20; x += 10) { ctx.beginPath(); ctx.moveTo(x, -25); ctx.lineTo(x + 5, -18); ctx.stroke(); }
  } else if (branch === "mortar:demolition") {
    ctx.strokeStyle = "#b7aaa0";
    ctx.beginPath(); ctx.arc(0, 0, 31, 0, Math.PI * 2); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(-31, 0); ctx.lineTo(31, 0); ctx.stroke();
  } else if (branch === "mortar:incendiary") {
    ctx.strokeStyle = "#e06a2f";
    ctx.beginPath(); ctx.moveTo(0, -30); ctx.lineTo(12, -12); ctx.lineTo(0, 2); ctx.lineTo(-12, -12); ctx.closePath(); ctx.stroke();
  } else if (branch === "spire:storm") {
    ctx.strokeStyle = "#d8d2c8";
    ctx.beginPath(); ctx.moveTo(-24, -18); ctx.lineTo(-8, -8); ctx.lineTo(-18, 2); ctx.lineTo(4, 14); ctx.lineTo(18, 2); ctx.stroke();
