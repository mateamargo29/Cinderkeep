  } else if (branch === "spire:frost") {
    ctx.strokeStyle = "#a9d8e8";
    for (let a = 0; a < Math.PI; a += Math.PI / 3) { ctx.beginPath(); ctx.moveTo(Math.cos(a) * 27, Math.sin(a) * 27); ctx.lineTo(-Math.cos(a) * 27, -Math.sin(a) * 27); ctx.stroke(); }
  }
}

function drawEnemies(ctx: CanvasRenderingContext2D, game: Game, assets: GameAssets) {
  const list = game.enemies.filter((e) => e.alive).sort((a, b) => a.y - b.y);
  for (const e of list) {
    const def = ENEMIES[e.kind];
    const atlas = assets[def.sprite];
    const frame = Math.floor(e.anim) % 4;
    const size = CELL * 0.85 * e.scale;
    ctx.save();
    ctx.translate(e.x, e.y);
    ctx.rotate(e.facing);
    const eraFilter =
      e.era === 0 ? "sepia(0.62) saturate(0.82) contrast(1.16) brightness(0.94)" :
      e.era === 1 ? "sepia(0.95) saturate(1.8) brightness(1.08)" :
      e.era === 2 ? "grayscale(0.72) contrast(1.28) brightness(1.08)" :
      e.era === 3 ? "grayscale(0.48) contrast(1.35) brightness(0.94)" :
      e.era === 4 ? "hue-rotate(155deg) saturate(1.8) brightness(1.22)" :
      "none";
    ctx.filter = e.flash > 0
      ? (eraFilter === "none" ? "brightness(1.8)" : `${eraFilter} brightness(1.55)`)
      : eraFilter;
    if (e.era === 4) {
      ctx.shadowBlur = e.boss ? 18 : 10;
      ctx.shadowColor = "#86e2ff";
    }
    drawFrame(ctx, atlas, frame, -size / 2, -size / 2, size, size);
    drawEnemyEvolution(ctx, e, size);
    if (e.fortifyTimer > 0) drawEnemyFortify(ctx, e, size);
    if (e.shield > 0) drawEnemyShield(ctx, e, size);
    if (e.abilityFlash > 0) drawBossAbilityAura(ctx, e, size);
    if (e.speedBoostTimer > 0) drawSpeedBoost(ctx, e, size);
    if (e.burn > 0) { ctx.strokeStyle = "rgba(224,106,47,0.9)"; ctx.lineWidth = 2; ctx.beginPath(); ctx.arc(0, 0, size * 0.56, 0, Math.PI * 2); ctx.stroke(); }
    ctx.filter = "none";
    ctx.restore();
    drawHp(ctx, e);
  }
}

function drawFrame(
  ctx: CanvasRenderingContext2D,
  atlas: Atlas,
  frame: number,
  x: number,
  y: number,
  w: number,
  h: number,
) {
  const fw = atlas.image.width / atlas.cols;
  const fh = atlas.image.height / atlas.rows;
  const col = frame % atlas.cols;
  const row = Math.floor(frame / atlas.cols);
  ctx.drawImage(atlas.image, col * fw, row * fh, fw, fh, x, y, w, h);
}

    ctx.beginPath(); ctx.arc(0, 0, size * 0.52, 0, Math.PI * 2); ctx.stroke();

function drawEnemyFortify(ctx: CanvasRenderingContext2D, e: Enemy, size: number) {
  ctx.save();
  ctx.globalAlpha = 0.38 + Math.min(0.3, e.fortify);
  ctx.strokeStyle = e.era >= 3 ? "#c45c3d" : "#d4a65b";
  ctx.lineWidth = 3;
  ctx.setLineDash([6, 5]);
  ctx.beginPath(); ctx.arc(0, 0, size * 0.62, 0, Math.PI * 2); ctx.stroke();
  ctx.restore();
}

function drawBossAbilityAura(ctx: CanvasRenderingContext2D, e: Enemy, size: number) {
  const colors = ["#d08a45", "#d4a65b", "#b8bdc1", "#c45c3d", "#86d6ef"];
  ctx.save();
  ctx.globalAlpha = Math.min(0.8, e.abilityFlash * 0.75);
  ctx.strokeStyle = colors[Math.max(0, Math.min(4, e.era))];
  ctx.lineWidth = e.boss ? 5 : 2.5;
  ctx.beginPath(); ctx.arc(0, 0, size * (0.7 + (1 - Math.min(1, e.abilityFlash)) * 0.18), 0, Math.PI * 2); ctx.stroke();
  ctx.restore();
}

function drawSpeedBoost(ctx: CanvasRenderingContext2D, e: Enemy, size: number) {
  ctx.save(); ctx.globalAlpha = 0.42; ctx.strokeStyle = e.era === 4 ? "#86d6ef" : "#d08a45"; ctx.lineWidth = 2;
  for (let i = -1; i <= 1; i++) { const y = i * 8; ctx.beginPath(); ctx.moveTo(-size * 0.58, y + 5); ctx.lineTo(-size * 0.72, y); ctx.lineTo(-size * 0.58, y - 5); ctx.stroke(); }
  ctx.restore();
}

function drawEnemyShield(ctx: CanvasRenderingContext2D, e: Enemy, size: number) {
  if (e.maxShield <= 0) return;
  const ratio = Math.max(0, Math.min(1, e.shield / e.maxShield));
  ctx.save();
  ctx.globalAlpha = 0.28 + ratio * 0.4;
  ctx.strokeStyle = "#a9d8e8";
  ctx.lineWidth = 2 + ratio * 2;
  ctx.beginPath();
  ctx.arc(0, 0, size * (0.58 + ratio * 0.05), 0, Math.PI * 2);
  ctx.stroke();
  ctx.restore();
}

function drawHp(ctx: CanvasRenderingContext2D, e: Enemy) {
  const w = 22 * e.scale + (e.boss ? 18 : 0);
  const x = e.x - w / 2;
  const y = e.y - 22 * e.scale;
  if (e.maxShield > 0) {
    ctx.fillStyle = "rgba(12,11,10,0.7)";
    ctx.fillRect(x, y - 5, w, 3);
    ctx.fillStyle = "#a9d8e8";
    ctx.fillRect(x, y - 5, w * Math.max(0, e.shield / e.maxShield), 3);
  }
  ctx.fillStyle = "rgba(12,11,10,0.7)";
  ctx.fillRect(x, y, w, 4);
  ctx.fillStyle = e.boss ? "#c45c26" : e.role === "armored" ? "#9b8c72" : "#6f8f6a";
  ctx.fillRect(x, y, w * Math.max(0, e.hp / e.maxHp), 4);
}

function drawProjectiles(ctx: CanvasRenderingContext2D, game: Game) {
  for (const p of game.projectiles) {
    if (!p.alive) continue;
    ctx.save();
    ctx.translate(p.x, p.y);
    const angle = Math.atan2(p.vy, p.vx);
    ctx.rotate(angle);
    if (p.techEra >= 2 || p.branchId) {
      const trail = p.techEra >= 4 ? 22 : p.techEra >= 3 ? 16 : 11;
      const trailColor =
        p.kind === "spire" ? (p.branchId === "spire:frost" ? "rgba(169,216,232,0.55)" : "rgba(240,192,138,0.48)") :
        p.kind === "mortar" ? (p.branchId === "mortar:incendiary" ? "rgba(224,106,47,0.48)" : "rgba(181,74,60,0.34)") :
        "rgba(216,210,200,0.38)";
      ctx.strokeStyle = trailColor;
      ctx.lineWidth = p.techEra >= 4 ? 3 : 2;
      ctx.beginPath();
      ctx.moveTo(-4, 0);
      ctx.lineTo(-trail, 0);
      ctx.stroke();
    }
    if (p.kind === "ballista") {
      if (p.techEra >= 4) {
        ctx.shadowBlur = 10;
        ctx.shadowColor = "#86d6ef";
      }
      ctx.fillStyle = p.branchId === "ballista:siege" ? "#d8c39b" : p.branchId === "ballista:rapid" ? "#eee8e0" : p.techEra >= 4 ? "#d8f3ff" : p.techEra >= 3 ? "#b54a3c" : p.techEra === 0 ? "#b7a77b" : "#c4b08a";
      const thickness = p.branchId === "ballista:siege" ? 4 : 2.8;
      ctx.fillRect(-8, -thickness / 2, 16, thickness);
      ctx.fillStyle = "#8a7a62";
      ctx.fillRect(6, -2.2, 5, 4.4);
    } else if (p.kind === "mortar") {
      if (p.techEra >= 4) {
        ctx.shadowBlur = 12;
        ctx.shadowColor = "#86d6ef";
      }
      ctx.fillStyle = p.branchId === "mortar:incendiary" ? "#e06a2f" : p.techEra >= 4 ? "#86d6ef" : p.techEra >= 3 ? "#8f8a82" : p.techEra === 0 ? "#625b4b" : "#2a2420";
      ctx.beginPath();
      ctx.arc(0, 0, p.branchId === "mortar:demolition" ? 7 : 5.5, 0, Math.PI * 2);
      ctx.fill();
      ctx.fillStyle = p.branchId === "mortar:incendiary" ? "#f0b06e" : "#c45c26";
      ctx.beginPath();
      ctx.arc(-1, -1, 2, 0, Math.PI * 2);
      ctx.fill();
    } else {
      ctx.shadowBlur = p.techEra >= 3 ? 12 : 7;
      ctx.shadowColor = p.branchId === "spire:frost" ? "#a9d8e8" : p.techEra >= 4 ? "#86d6ef" : "#f0c08a";
      const g = ctx.createRadialGradient(0, 0, 1, 0, 0, p.techEra >= 4 ? 12 : 9);
      g.addColorStop(0, p.branchId === "spire:frost" ? "#d8f3ff" : "#f0c08a");
      g.addColorStop(1, p.branchId === "spire:frost" ? "rgba(108,188,224,0)" : "rgba(196,92,38,0)");
      ctx.fillStyle = g;
      ctx.beginPath();
      ctx.arc(0, 0, 9, 0, Math.PI * 2);
      ctx.fill();
    }
    ctx.restore();
  }
}

function drawParticles(ctx: CanvasRenderingContext2D, game: Game) {
  for (const p of game.particles) {
    if (!p.alive) continue;
    ctx.globalAlpha = Math.max(0, p.life / p.maxLife);
    ctx.fillStyle = p.color;
    ctx.beginPath();
    ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
    ctx.fill();
    ctx.globalAlpha = 1;
  }
}

function drawFloats(ctx: CanvasRenderingContext2D, game: Game) {
  ctx.font = "600 12px Figtree, sans-serif";
  ctx.textAlign = "center";
  for (const f of game.floats) {
    if (!f.alive) continue;
    ctx.globalAlpha = Math.max(0, f.life / 0.7);
    ctx.fillStyle = f.color;
    ctx.fillText(f.text, f.x, f.y);
    ctx.globalAlpha = 1;
  }
}

  ctx.save();

    for (let i = 0; i < 18; i++) { const x = (i * 83 + Math.sin(i * 3.1) * 30) % WORLD_W; const y = WORLD_H - ((time * 26 + i * 57) % WORLD_H); ctx.fillStyle = `rgba(214,91,45,${0.18 + (i % 4) * 0.05})`; ctx.beginPath(); ctx.arc(x, y, 1.2 + (i % 3) * 0.5, 0, Math.PI * 2); ctx.fill(); }

function drawVignette(ctx: CanvasRenderingContext2D) {
  const g = ctx.createRadialGradient(
    WORLD_W / 2,
    WORLD_H / 2,
    WORLD_H * 0.2,
    WORLD_W / 2,
    WORLD_H / 2,

  g.addColorStop(1, "rgba(8,7,6,0.45)");
  ctx.fillStyle = g;
  ctx.fillRect(0, 0, WORLD_W, WORLD_H);
}
